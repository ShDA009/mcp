# Единственное место, где фиксируются версии MCP-серверов, запускаемых
# без Docker.
#
# Лаунчеры (~/.config/<server>/launch.sh, launch.ps1 на Windows) подтягивают
# этот файл при каждом старте сервера с raw.githubusercontent.com — правка
# здесь + push доезжает до всех сотрудников без переустановки
# (см. <server>/install/README.md).
#
# --- Сторонние пакеты (uvx/npx, ставятся из PyPI/npm) -----------------------
# Патчи внутри диапазона (например 0.23.0 -> 0.23.4) подхватываются сами.
# Подъём на новую минорную/мажорную ветку — правка строки ниже + коммит.

ATLASSIAN_SPEC="mcp-atlassian>=0.23,<0.24"
GITLAB_SPEC="@zereight/mcp-gitlab@^2.1"

# --- Наши серверы (venv, ставятся из этого репозитория) ---------------------
# Git-ref (ветка, тег или коммит), который лаунчер ставит в локальный venv.
# Лаунчер сверяет это значение с venv\.installed-ref и переустанавливает
# пакет ТОЛЬКО при расхождении — обычный старт не резолвит зависимости.
#
# Раньше эти два сервера запускались как `uvx --from git+...` на каждый старт;
# под Cline это давало "Failed to resolve --with requirement" / Connection
# closed, потому что резолв из git выполнялся в горячем пути запуска.
#
# Для отката на конкретную версию поставьте тег или SHA коммита вместо master.

OUTLOOK_REF="master"
ZEPHYR_REF="master"
